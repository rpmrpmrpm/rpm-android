# Media3's own rules for the FFmpeg extension: the JNI methods, and the renderer's constructor, which
# DefaultRenderersFactory finds by reflection.
-keepclasseswithmembernames class * {
    native <methods>;
}
# JNI_OnLoad looks up FfmpegAudioDecoder.growOutputBuffer by name. Nothing in Java calls it, so without this
# R8 strips it from minified builds, JNI_OnLoad fails and FFmpeg never loads.
-keep class androidx.media3.decoder.ffmpeg.FfmpegAudioDecoder {
    *;
}
-keepclassmembers class androidx.media3.decoder.ffmpeg.FfmpegAudioRenderer {
    <init>(android.os.Handler, androidx.media3.exoplayer.audio.AudioRendererEventListener, androidx.media3.exoplayer.audio.AudioSink);
}
